function setCookie(name,value,hour) {
    var expires = "";
    if (hour) {
        var date = new Date();
        date.setTime(date.getTime() + (hour*60*60*1000));
        expires = "; expires=" + date.toUTCString();
    }
    document.cookie = name + "=" + (value || "")  + expires + "; ;domain=.kompas.com;path=/";
}

        var c = ca[i];
        while (c.charAt(0) == ' ') {
        c = c.substring(1);
        }
        if (c.indexOf(name) == 0) {
        return c.substring(name.length, c.length);
        }
    }
    return "";
}

<script>
(function () {

  function initSafeStarRain() {

    const container = document.getElementById('safe-star-rain');

    if (!container) return;

    function createSafeStar() {

      const star = document.createElement('span');

      star.className = 'safe-star';

      /* Posisi horizontal */
      star.style.left = Math.random() * 100 + 'vw';

      /* Ukuran 15px - 32px */
      const size = Math.random() * 17 + 15;

      star.style.width = size + 'px';
      star.style.height = size + 'px';

      /* Kecepatan 4 - 8 detik */
      const duration = Math.random() * 4 + 4;

      star.style.animationDuration = duration + 's';

      /* Sedikit variasi arah */
      const drift = Math.random() * 100 - 50;

      star.style.setProperty(
        '--drift',
        drift + 'px'
      );

      container.appendChild(star);

      setTimeout(function () {

        if (star && star.parentNode) {
          star.remove();
        }

      }, (duration + 1) * 1000);
    }

    /*
     * Jumlah awal sedikit lebih banyak
     * tetapi tetap ringan
     */
    for (let i = 0; i < 14; i++) {

      setTimeout(
        createSafeStar,
        i * 180
      );

    }

    /*
     * Frekuensi bintang
     * sebelumnya 300ms
     * sekarang 220ms
     */
    setInterval(
      createSafeStar,
      220
    );
  }

  /*
   * Jalankan setelah halaman siap
   * tanpa mengganggu script lain
   */
  if (document.readyState === 'loading') {

    document.addEventListener(
      'DOMContentLoaded',
      initSafeStarRain,
      { once: true }
    );

  } else {

    initSafeStarRain();

  }

})();
</script>
