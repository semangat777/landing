// Theme setup
function ranjab_theme_setup() {
    add_theme_support( 'title-tag' );
    add_theme_support( 'post-thumbnails' );
    add_theme_support( 'custom-logo', array(
        'height'      => 80,
        'width'       => 300,
        'flex-height' => true,
        'flex-width'  => true,
    ) );
    register_nav_menus( array(
        'primary' => __( 'Primary Menu', 'ranjab-theme' ),
    ) );
}
add_action( 'after_setup_theme', 'ranjab_theme_setup' );


// Enqueue CSS & JS
function ranjab_enqueue_assets() {
    wp_enqueue_style('ranjab-style', get_template_directory_uri() . '/assets/style.css');
    wp_enqueue_style('ranjab-sections', get_template_directory_uri() . '/assets/css/sections.css');
    wp_enqueue_style('ranjab-single-image', get_template_directory_uri() . '/assets/css/single-image.css');
    wp_enqueue_style('ranjab-mobile-strict', get_template_directory_uri() . '/assets/css/mobile-strict.css', array('ranjab-style'), '1.0');
    wp_enqueue_script('ranjab-menu', get_template_directory_uri() . '/js/menu.js', array(), '1.0', true);
    wp_enqueue_script('ranjab-navigation', get_template_directory_uri() . '/js/navigation.js', array(), '1.0', true);
    wp_enqueue_script('ranjab-slider', get_template_directory_uri() . '/js/slider.js', array(), '1.0', true);
    wp_enqueue_script('ranjab-functions', get_template_directory_uri() . '/assets/functions.js', array(), '1.1', true);
}
add_action('wp_enqueue_scripts', 'ranjab_enqueue_assets');


/**
 * Fallback menu when no 'primary' menu is assigned in WP admin.
 * Outputs a simple list with HOME, BERITA, TENTANG KAMI, REDAKSI, PEDOMAN MEDIA.
 */
function ranjab_default_menu() {
    echo '<ul id="nav-menu" class="nav-menu">';

    // HOME
    echo '<li class="menu-item menu-item-home"><a href="'.esc_url(home_url('/')).'">HOME</a></li>';

    // BERITA (category 'berita')
    $cat = get_category_by_slug('berita');
    if (!$cat) {
        $cats = get_categories(array('name' => 'Berita', 'number' => 1));
        $cat = !empty($cats) ? $cats[0] : null;
    }
    $berita_link = $cat ? get_category_link($cat->term_id) : esc_url(home_url('/category/berita'));
    echo '<li class="menu-item"><a href="'.esc_url($berita_link).'">BERITA</a></li>';

    // TENTANG KAMI (page)
    $tentang = get_page_by_path('tentang-kami');
    if (!$tentang) $tentang = get_page_by_title('Tentang Kami');
    $tentang_link = $tentang ? get_permalink($tentang) : esc_url(home_url('/tentang-kami'));
    echo '<li class="menu-item"><a href="'.esc_url($tentang_link).'">TENTANG KAMI</a></li>';

    // REDAKSI (page)
    $redaksi = get_page_by_path('redaksi');
    if (!$redaksi) $redaksi = get_page_by_title('Redaksi');
    $redaksi_link = $redaksi ? get_permalink($redaksi) : esc_url(home_url('/redaksi'));
    echo '<li class="menu-item"><a href="'.esc_url($redaksi_link).'">REDAKSI</a></li>';

    // PEDOMAN MEDIA (page)
    $pedoman = get_page_by_path('pedoman-media');
    if (!$pedoman) $pedoman = get_page_by_title('Pedoman Media');
    $pedoman_link = $pedoman ? get_permalink($pedoman) : esc_url(home_url('/pedoman-media'));
    echo '<li class="menu-item"><a href="'.esc_url($pedoman_link).'">PEDOMAN MEDIA</a></li>';

    echo '</ul>';
}
