class FreeAPIs:
    @staticmethod
    def get_google_api_key():

        """Rotate CSE IDs"""

# --- SEARCH ENGINES (Enhanced) ---
class SearchEngines:
    @staticmethod
    def google_api(query, tld, proxy=None):
        """Google Custom Search API - Most reliable"""
        api_key = FreeAPIs.get_google_api_key()
        cse_id = FreeAPIs.get_cse_id()
        url = f"https://www.googleapis.com/customsearch/v1?key={api_key}&cx={cse_id}&q={quote_plus(query)}&num=10"
        
        for retry in range(2):
            try:
                headers = {'User-Agent': random.choice(UA_LIST)}
                r = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT, proxies=proxy, verify=False)
                
                if r.status_code == 200:
                    data = r.json()
                    domains = []
                    for item in data.get('items', []):
                        match = re.search(r'https?://([a-zA-Z0-9.-]+\.' + tld + r')', item.get('link', ''))
                        if match:
                            domains.append(match.group(1))
                    return domains
                elif r.status_code == 429:  # Rate limit, coba key lain
                    time.sleep(2)
                    continue
            except:
                if retry == 0:
                    time.sleep(1)
        return []
    
    @staticmethod
    def bing_mobile(query, tld, proxy=None):
        """Bing dengan mobile user agent"""
        url = f"https://www.bing.com/search?q={quote_plus(query)}&count=50&mkt=en-US"
        try:
            # Pake mobile UA
            headers = {'User-Agent': random.choice([ua for ua in UA_LIST if 'iPhone' in ua or 'Android' in ua])}
            r = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT, proxies=proxy, verify=False)
            pattern = r'https?://([a-zA-Z0-9.-]+\.' + tld + r')(?:/|$|\s|")'
            results = re.findall(pattern, r.text)
            # Filter out search engine domains
            return [d for d in results if 'bing' not in d and 'microsoft' not in d]
        except:
            return []
    
    @staticmethod
    def bing_regular(query, tld, proxy=None):
        """Bing regular desktop"""
        url = f"https://www.bing.com/search?q={quote_plus(query)}&count=50"
        try:
            headers = {'User-Agent': random.choice(UA_LIST)}
            r = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT, proxies=proxy, verify=False)
            pattern = r'https?://([a-zA-Z0-9.-]+\.' + tld + r')(?:/|$|\s|")'
            results = re.findall(pattern, r.text)
            return [d for d in results if 'bing' not in d and 'microsoft' not in d]
        except:
            return []
    
    @staticmethod
    def duckduckgo_mobile(query, tld, proxy=None):
        """DuckDuckGo Mobile version"""
        url = f"https://duckduckgo.com/html/?q={quote_plus(query)}"
        try:
            headers = {'User-Agent': random.choice([ua for ua in UA_LIST if 'Mobile' in ua or 'iPhone' in ua])}
            r = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT, proxies=proxy, verify=False)
            pattern = r'https?://([a-zA-Z0-9.-]+\.' + tld + r')'
            return re.findall(pattern, r.text)
        except:
            return []
    
    @staticmethod
    def duckduckgo_lite(query, tld, proxy=None):
        """DuckDuckGo Lite - No JS"""
        url = f"https://lite.duckduckgo.com/lite/?q={quote_plus(query)}"
        try:
            headers = {'User-Agent': random.choice(UA_LIST)}
            r = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT, proxies=proxy, verify=False)
            pattern = r'https?://([a-zA-Z0-9.-]+\.' + tld + r')'
            return re.findall(pattern, r.text)
        except:
            return []
    
    @staticmethod
    def qwant_mobile(query, tld, proxy=None):
        """Qwant with mobile optimization"""
        url = f"https://api.qwant.com/v3/search/web?q={quote_plus(query)}&count=15&device=mobile"
        try:
            headers = {
                'User-Agent': random.choice([ua for ua in UA_LIST if 'Mobile' in ua]),
                'Accept': 'application/json'
            }
            r = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT, proxies=proxy, verify=False)
            data = r.json()
            domains = []
            for item in data.get('data', {}).get('result', {}).get('items', []):
                match = re.search(r'https?://([a-zA-Z0-9.-]+\.' + tld + r')', item.get('url', ''))
                if match:
                    domains.append(match.group(1))
            return domains
        except:
            return []
    
    @staticmethod
    def yandex_mobile(query, tld, proxy=None):
        """Yandex mobile version"""
        url = f"https://yandex.com/touch/search?text={quote_plus(query)}"
        try:
            headers = {'User-Agent': random.choice([ua for ua in UA_LIST if 'Mobile' in ua])}
            r = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT, proxies=proxy, verify=False)
            pattern = r'https?://([a-zA-Z0-9.-]+\.' + tld + r')'
            return re.findall(pattern, r.text)
        except:
            return []

class SonicGacorV10:
    def __init__(self, tlds):
        if isinstance(tlds, str):
            self.tlds = [tlds.strip('.')]
        else:
            self.tlds = [t.strip('.') for t in tlds if t]
        
        self.found_domains = set()
        self.print_lock = Lock()
        self.proxy_index = 0
        self.api_key_index = 0
        self.results_history = []
        
    def get_proxy(self):
        """Rotating proxy"""
        self.proxy_index = (self.proxy_index + 1) % len(FREE_PROXIES)
        proxy = FREE_PROXIES[self.proxy_index]
        if proxy:
            return {'http': proxy, 'https': proxy}
        return None
    
    def safe_print(self, msg):
        with self.print_lock:
            print(msg)
    
    def search_all_engines(self, query, tld):
        """Multi-engine dengan multiple variants"""
        engines = [
            ('Google API', SearchEngines.google_api),
            ('Bing Mobile', SearchEngines.bing_mobile),
            ('Bing Desktop', SearchEngines.bing_regular),
            ('DDG Mobile', SearchEngines.duckduckgo_mobile),
            ('DDG Lite', SearchEngines.duckduckgo_lite),
            ('Qwant Mobile', SearchEngines.qwant_mobile),
            ('Yandex Mobile', SearchEngines.yandex_mobile),
        ]
        
        all_domains = []
        
        for engine_name, engine_func in engines:
            for retry in range(MAX_RETRIES):
                proxy = self.get_proxy()
                proxy_str = proxy['http'] if proxy else 'Direct'
                
                self.safe_print(f"  [*] {engine_name} via {proxy_str} (try {retry+1})")
                
                domains = engine_func(query, tld, proxy)
                
                if domains:
                    self.safe_print(f"  [+] {engine_name} found {len(domains)} domains")
                    all_domains.extend(domains)
                    
                    # Save sample results
                    if domains and len(self.results_history) < 10:
                        self.results_history.append({
                            'engine': engine_name,
                            'query': query,
                            'count': len(domains),
                            'sample': domains[:3]
                        })
                    break
                else:
                    if retry < MAX_RETRIES - 1:
                        time.sleep(random.uniform(1, 2))
            
            # Jeda antar engine
            time.sleep(random.uniform(0.5, 1))
        
        return list(set(all_domains))
    
    def dork_for_tld(self, tld):
        """Generate dorks untuk satu TLD"""
        self.safe_print(f"\n{'='*60}")
        self.safe_print(f"[*] SCANNING TLD: .{tld}")
        self.safe_print(f"{'='*60}")
        
        # Expanded dorks
        dorks = [
            # OJS-specific
            f'site:{tld} "Open Journal Systems"',
            f'site:{tld} "OJS" "journal" "pkp"',
            f'site:{tld} inurl:"index.php/index/login"',
            f'site:{tld} inurl:"/journals/"',
            
            # Path-based
            f'site:{tld} "/files/journals/"',
            f'site:{tld} "/ojsfiles/journals/"',
            f'site:{tld} "/ojsdata/journals/"',
            
            # Version detection
            f'site:{tld} "pkp" "ojs2"',
            f'site:{tld} "pkp" "ojs3"',
            
            # Generic journal
            f'site:{tld} "journal" "current issue"',
            f'site:{tld} "vol" "no" "journal"',
            
            # Include variants
            f'site:{tld} intitle:"journal" inurl:"index"',
            f'site:{tld} "issn" "journal"',
        ]
        
        tld_domains = set()
        
        for idx, dork in enumerate(dorks, 1):
            self.safe_print(f"\n[>] Dork {idx}/{len(dorks)}: {dork[:80]}")
            domains = self.search_all_engines(dork, tld)
            
            if domains:
                self.safe_print(f"  [✓] Got {len(domains)} unique domains")
                tld_domains.update(domains)
            else:
                self.safe_print(f"  [✗] No results")
            
            # Dynamic delay
            delay = random.uniform(1.5, 3.5)
            if idx % 3 == 0:
                delay += 2  # Extra delay setiap 3 dorks
            self.safe_print(f"  [⏱] Waiting {delay:.1f}s...")
            time.sleep(delay)
        
        for dom in tld_domains:
            self.found_domains.add(f"http://{dom}")
        
        self.safe_print(f"\n[+] TLD .{tld} complete: {len(tld_domains)} domains found")
        return tld_domains
    
    def verify_path(self, domain, path):
        """Verifikasi path dengan multiple retry dan proxy rotation"""
        target = f"{domain.rstrip('/')}/{path}"
        
        for attempt in range(MAX_RETRIES):
            proxy = self.get_proxy()
            headers = {'User-Agent': random.choice(UA_LIST)}
            
            try:
                r = requests.get(
                    target, 
                    headers=headers, 
                    timeout=REQUEST_TIMEOUT, 
                    verify=False, 
                    allow_redirects=False,
                    proxies=proxy
                )
                
                if r.status_code in [200, 403, 404]:
                    return f"{target} [{r.status_code}]"
                    
            except requests.exceptions.ProxyError:
                if attempt < MAX_RETRIES - 1:
                    time.sleep(1)
                    continue
            except:
                if attempt < MAX_RETRIES - 1:
                    time.sleep(random.uniform(0.5, 1))
                    continue
                break
                
        return None
    
    def validate_all_domains(self):
        """Multi-thread validation"""
        if len(self.found_domains) == 0:
            self.safe_print("\n[-] No domains found.")
            self.safe_print("\n[!] TROUBLESHOOTING:")
            self.safe_print("   1. Coba ganti API keys di GOOGLE_API_KEYS (daftar gratis di Google Cloud)")
            self.safe_print("   2. Gunakan VPN atau ganti IP")
            self.safe_print("   3. Coba TLD yang lebih besar: .com, .net, .org")
            self.safe_print("   4. Tunggu 5-10 menit lalu coba lagi")
            self.safe_print("\n[+] Alternative: Jalankan dengan TLD .com dulu:")
            self.safe_print("   python sonic.py com")
            return
        
        self.safe_print(f"\n[*] Validating paths on {len(self.found_domains)} domains...")
        self.safe_print(f"[*] Using {THREADS} threads\n")
        
        total_tasks = len(self.found_domains) * len(TARGET_PATHS)
        completed = 0
        valid_count = 0
        
        with ThreadPoolExecutor(max_workers=THREADS) as executor:
            futures = [executor.submit(self.verify_path, dom, path) 
                      for dom in self.found_domains for path in TARGET_PATHS]
            
            for future in as_completed(futures):
                completed += 1
                res = future.result()
                
                if res:
                    valid_count += 1
                    with self.print_lock:
                        print(f" [✓] VALID #{valid_count}: {res}")
                        with open(OUTPUT, "a") as f_out:
                            f_out.write(res + "\n")
                
                if completed % 100 == 0:
                    self.safe_print(f"[*] Progress: {completed}/{total_tasks} | Valid: {valid_count}")
    
    def save_report(self):
        """Save detailed report"""
        report_file = f"report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
        with open(report_file, 'w') as f:
            f.write("SONIC GACOR V10.0 - SCAN REPORT\n")
            f.write("="*50 + "\n\n")
            f.write(f"TLDs Scanned: {', '.join(self.tlds)}\n")
            f.write(f"Total Domains Found: {len(self.found_domains)}\n")
            f.write(f"Output File: {OUTPUT}\n\n")
            
            if self.results_history:
                f.write("Sample Results from Search Engines:\n")
                for res in self.results_history[:5]:
                    f.write(f"  - {res['engine']}: {res['count']} domains from '{res['query'][:50]}'\n")
                    if res.get('sample'):
                        f.write(f"    Samples: {', '.join(res['sample'][:3])}\n")
        
        self.safe_print(f"\n[+] Report saved: {report_file}")
    
    def start(self):
        """Main execution"""
        start_time = time.time()
        
        print("""
╔═══════════════════════════════════════════════════════════════╗
║     SONIC GACOR V10.0 - ULTIMATE EDITION                     ║
║     Google API + Mobile UA + Multi-Engine + Anti-Gosong     ║
╚═══════════════════════════════════════════════════════════════╝
        """)
        
        print(f"[*] Target TLDs: {', '.join(self.tlds)}")
        print(f"[*] Output file: {OUTPUT}")
        print(f"[*] Using {len(FREE_PROXIES)} proxies")
        print(f"[*] Using {len(GOOGLE_API_KEYS)} Google API keys")
        print(f"[*] Using {len(UA_LIST)} User Agents (Mobile + Desktop)")
        print()
        
        # Scan each TLD
        for idx, tld in enumerate(self.tlds, 1):
            self.safe_print(f"\n[>>] Processing TLD {idx}/{len(self.tlds)}: .{tld}")
            self.dork_for_tld(tld)
            
            if idx < len(self.tlds):
                delay = 5
                self.safe_print(f"\n[⏱] Waiting {delay}s before next TLD...")
                time.sleep(delay)
        
        print(f"\n{'='*60}")
        print(f"[+] TOTAL UNIQUE DOMAINS: {len(self.found_domains)}")
        
        # Validate
        if self.found_domains:
            self.validate_all_domains()
            self.save_report()
        else:
            print("\n[!] NO RESULTS - EMERGENCY MODE:")
            print("   Solusi cepat:")
            print("   1. Daftar Google API Key gratis: https://developers.google.com/custom-search/v1/introduction")
            print("   2. Atau pake VPN lalu coba lagi")
            print("   3. Atau ganti TLD ke .com dulu: python sonic.py com")
            print("\n   Demo dengan TLD .com (testing):")
            print("   python sonic.py com")
        
        elapsed = time.time() - start_time
        print(f"\n[DONE] Time: {elapsed:.1f}s ({elapsed/60:.1f} minutes)")
        print(f"[+] Results saved to: {OUTPUT}")
