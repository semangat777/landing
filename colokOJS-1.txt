if (!function_exists('stratix_setup')):
	/**
	 * Sets up theme defaults and registers support for various WordPress features.
	 *
	 * Note that this function is hooked into the after_setup_theme hook, which
	 * runs before the init hook. The init hook is too late for some features, such
	 * as indicating support for post thumbnails.
	 */
	function stratix_setup()
	{
		/*
		 * Make theme available for translation.
		 * Translations can be filed in the /languages/ directory.
		 * If you're building a theme based on Stratix, use a find and replace
		 * to change 'stratix' to the name of your theme in all the template files.
		 */
		load_theme_textdomain('stratix', get_template_directory() . '/languages');
	}
endif; 
add_action('after_setup_theme', 'stratix_setup');

if (!function_exists('stratix_enqueue')):
	/**
	 * Enqueue theme styles and scripts
	 *
	 * @since 1.0
	 * @return void
	 */
	function stratix_enqueue()
	{
		// rtl style
		wp_enqueue_style('stratix-style', get_stylesheet_uri(), array(), wp_get_theme()->get('Version'));
		wp_style_add_data('stratix-style', 'rtl', 'replace');

		// enqueue script
		wp_enqueue_script('stratix-custom', get_template_directory_uri() . '/assets/js/custom.js', array(), '', true);
	}
endif;
add_action('wp_enqueue_scripts', 'stratix_enqueue');

if (!function_exists('stratix_body_classes')):
	/**
	 * Adds custom classes to the array of body classes.
	 *
	 * @param array $classes Classes for the body element.
	 * @return array
	 */
	function stratix_body_classes($classes)
	{

		$classes[] = get_stylesheet();

		$theme = wp_get_theme();
		$classes[] = $theme->get('TextDomain');

		return $classes;
	}
endif;
add_filter('body_class', 'stratix_body_classes');

/**
 * Registers block patterns and categories.
 */
if (!function_exists('stratix_register_block_patterns')):
	/**
	 * Register pattern categories
	 *
	 * @since 1.0
	 * @return void
	 */
	function stratix_register_block_patterns()
	{

		register_block_pattern_category(
			'stratix',
			array(
				'label' => _x('Stratix', 'Block pattern category', 'stratix'),
			)
		);
	}
endif;

add_action('init', 'stratix_register_block_patterns');
