<?php
// Versi Lite - Biar gak 503
set_time_limit(60); // Kasih waktu 1 menit
$searchDir = '/home/u268132155/domains/citizen-science-1ocean.zooreach.org/public_html/';
$searchFor = 'noindex';

echo "<h3>Searching in: $searchDir</h3><pre>";

// Filter ekstensi biar gak berat (gak perlu scan gambar/zip)
$allowed_extensions = ['php', 'html', 'htm', 'js', 'inc', 'tpl'];

try {
    $directory = new RecursiveDirectoryIterator($searchDir);
    $iterator = new RecursiveIteratorIterator($directory);
    
    foreach ($iterator as $info) {
        $ext = pathinfo($info->getFilename(), PATHINFO_EXTENSION);
        
        if ($info->isFile() && in_array(strtolower($ext), $allowed_extensions)) {
            // Baca file baris demi baris biar gak makan RAM (Memory Efficient)
            $handle = fopen($info->getPathname(), "r");
            $line_num = 1;
            if ($handle) {
                while (($line = fgets($handle)) !== false) {
                    if (strpos($line, $searchFor) !== false) {
                        echo "<b>KETEMU:</b> " . $info->getPathname() . "\n";
                        echo "Baris $line_num: " . htmlspecialchars(trim($line)) . "\n";
                        echo "--------------------------------------------------\n";
                        break; // Ketemu satu di file ini, lanjut file berikutnya
                    }
                    $line_num++;
                }
                fclose($handle);
            }
        }
    }
} catch (Exception $e) {
    echo "Error: " . $e->getMessage();
}

echo "Scan Selesai.</pre>";
?>
